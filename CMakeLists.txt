cmake_minimum_required(VERSION 3.22)
project(eqt VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS eqt_compile=2 eqt_link=1)
set(CMAKE_JOB_POOL_COMPILE eqt_compile)
set(CMAKE_JOB_POOL_LINK eqt_link)
option(EQT_CPU_VARIANTS "Build GGML CPU backends for each Android ISA level and select one at runtime" ${ANDROID})
if(CMAKE_LIBRARY_OUTPUT_DIRECTORY)
    # Android Gradle packages only this directory; GGML writes dynamic backends to the runtime directory.
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
else()
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(EQT_CPU_VARIANTS)
    if(NOT ANDROID)
        message(FATAL_ERROR "EQT_CPU_VARIANTS uses GGML's Android variant list")
    endif()
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
endif()
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES
        LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_DIR}/include/llama.h")
    message(FATAL_ERROR "Run scripts/bootstrap.ps1 first")
endif()
add_subdirectory("${LLAMA_DIR}" llama)
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_library(eqt_template STATIC
    ${LLAMA_DIR}/common/json.cpp
    ${LLAMA_DIR}/common/unicode.cpp
    ${LLAMA_DIR}/common/jinja/lexer.cpp
    ${LLAMA_DIR}/common/jinja/parser.cpp
    ${LLAMA_DIR}/common/jinja/runtime.cpp
    ${LLAMA_DIR}/common/jinja/value.cpp
    ${LLAMA_DIR}/common/jinja/string.cpp)
target_include_directories(eqt_template PUBLIC ${LLAMA_DIR}/common PRIVATE ${LLAMA_DIR}/src ${LLAMA_DIR}/vendor)
target_link_libraries(eqt_template PUBLIC llama)
add_library(eqt_simd STATIC native/simd.cpp)
target_include_directories(eqt_simd PUBLIC native)
add_library(eqt_engine STATIC native/engine.cpp native/expert_store.cpp)
target_include_directories(eqt_engine PUBLIC native ${LLAMA_DIR}/vendor PRIVATE ${LLAMA_DIR}/ggml/src ${LLAMA_DIR}/src)
target_link_libraries(eqt_engine PUBLIC eqt_simd)
target_link_libraries(eqt_engine PUBLIC eqt_template)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE EQT_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE EQT_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
if(EQT_DIRTY)
    string(APPEND EQT_REV "-dirty")
endif()
target_compile_definitions(eqt_engine PRIVATE EQT_REV="${EQT_REV}" EQT_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
if(MSVC)
    target_compile_options(eqt_engine PRIVATE /W4 /utf-8)
else()
    target_compile_options(eqt_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(eqt-bench native/bench.cpp)
target_link_libraries(eqt-bench PRIVATE eqt_engine)
add_executable(eqt-check native/check.cpp)
target_link_libraries(eqt-check PRIVATE eqt_engine)
add_executable(eqt-kernel-bench native/kernel_bench.cpp)
target_link_libraries(eqt-kernel-bench PRIVATE eqt_engine)
add_executable(eqt-simd-test native/simd_test.cpp)
target_link_libraries(eqt-simd-test PRIVATE eqt_simd)
if(ANDROID)
    # Static so the same binary also runs under qemu-aarch64 without Android's dynamic linker.
    target_link_options(eqt-simd-test PRIVATE -static)
endif()
if(ANDROID)
    target_link_libraries(eqt_engine PUBLIC log)
    add_executable(eqt-io-bench native/io_bench.cpp)
    target_link_libraries(eqt-io-bench PRIVATE eqt_engine)
    add_library(eqt_jni SHARED native/jni.cpp)
    target_link_libraries(eqt_jni PRIVATE eqt_engine log)
endif()
